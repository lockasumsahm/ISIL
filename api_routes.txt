from fastapi import APIRouter

from app import __version__
from app.adapters import build_adapters
from app.api.schemas.response import HealthResponse
from app.config import get_settings

router = APIRouter(tags=["health"])


@router.get("/health", response_model=HealthResponse)
async def health() -> HealthResponse:
    settings = get_settings()
    adapters = build_adapters(settings)
    status_map = {}
    for a in adapters:
        name = a.name
        if name == "perspective":
            status_map[name] = "live" if settings.has_perspective() else "mock"
        elif name == "openai":
            status_map[name] = "live" if settings.has_openai() else "mock"
        elif name == "huggingface":
            status_map[name] = "live" if settings.has_huggingface() else "mock"
        else:
            status_map[name] = "mock"
    return HealthResponse(status="ok", version=__version__, adapters=status_map)

from pathlib import Path

from fastapi import APIRouter
from fastapi.responses import HTMLResponse

router = APIRouter(tags=["dashboard"])

DASHBOARD_HTML = Path(__file__).resolve().parent.parent / "static" / "dashboard.html"


@router.get("/dashboard", response_class=HTMLResponse)
async def dashboard():
    if DASHBOARD_HTML.exists():
        return DASHBOARD_HTML.read_text()
    return "<h1>ISIL Dashboard</h1><p>dashboard.html not found</p>"
